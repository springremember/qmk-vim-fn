# engine/test/mutants/mutants.txt —— 声明式变异记录表（mutation gate）
#
# 每条记录 = 一次「把实现改坏」的整行文本替换；runner（run.py）施加替换、跑套件、
# 分类 CAUGHT / SURVIVED，然后**无条件还原**。
#
# 格式（人类可编辑、diff 友好；不用 JSON 是因为 old/new 里全是多行 C 代码，
# 转义成 \n 会毁掉可读性与评审价值）：
#
#   mutant <id>          记录开始；id 稳定、全局唯一
#     file  = <仓库相对路径>
#     suite = engine | glue        （engine = make test；glue = make glue-test）
#     equiv = no | yes             （yes 表示已知等价变异，SURVIVED 不算失败）
#     why   = <一句话>              （必填：破坏什么行为；equiv=yes 时是等价性论证）
#     old
#       <逐行原文，必须与当前源码逐字节一致且唯一匹配，否则 drift 报错>
#     new
#       <替换后的逐行文本；允许为空（删除整段）>
#     end
#
# 语义：每个块里的一行都自带结尾换行，因此补丁**永远是整行替换**，不会吃掉/留下半行。
# 加一条变异 = 复制一组 `mutant … end` 填进去（见 qmk/engineering-spec.md §1.1）。

mutant budget-room-unbounded
file = engine/src/command.c
suite = engine
equiv = no
why = 键码预算失效：room 恒为 1<<20 ⇒ 所有带计数发射器都不再截断，队列被撑到 EMIT_CAP=256 后静默丢键（命令半截执行 = 数据损坏）。
old
int kv_emit_room(void) {
    int room = KV_CMD_KEY_BUDGET - kv_emit_pending();
    return (room > 0) ? room : 0;
}
new
int kv_emit_room(void) {
    return 1 << 20;   /* 变异：假装预算永远够 */
}
end

mutant budget-clamp-n-neutered
file = engine/src/command.c
suite = engine
equiv = no
why = kv_emit_clamp_n 不再按剩余预算截断计数 ⇒ 预填队列后连发大计数命令会到顶丢键（不变式 b 破）。
old
int kv_emit_clamp_n(int n, int fixed, int per) {
    if (n < 1) n = 1;
    if (per < 1) per = 1;
    int maxn = (kv_emit_room() - fixed) / per;
    if (maxn < 1) maxn = 1;
    return (n > maxn) ? maxn : n;
}
new
int kv_emit_clamp_n(int n, int fixed, int per) {
    if (n < 1) n = 1;
    if (per < 1) per = 1;
    (void)fixed;
    return n;   /* 变异：不截断 */
}
end

mutant d23-vline-esc-cursor

old

                /* 预算（design §4.4）：Up×off 至多 100 键；这里可以**少发到 0**（不是带计数
                 * 命令的"至少 1 次"），预算里预扣随后的 Home 与 Esc 两个固定键。 */

                kv_emit_tap(KV_HOME);
            }

            /* 变异：D23 行可视 Esc 不回选区起点行首 */
end
